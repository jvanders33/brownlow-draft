function render(){
  if(!state)return;

function renderSignup(){
  const list=$('signup-list');list.innerHTML='';
  state.signups.forEach((name,i)=>{
    const li=document.createElement('li');
    li.innerHTML=`<span class="pick-num">${i+1}</span><span class="name">${esc(name)}</span>`;
    list.appendChild(li);
  });

  const now=new Date(), deadline=new Date(now);

function renderLeaderboard(){

function getTotals(){
  const t={};
  Object.entries(state.picks).forEach(([team,players])=>{
    t[team]={total:0,players};
    players.forEach(p=>{let pt=0;Object.values(state.votes).forEach(rv=>pt+=(rv[p]||0));t[team].total+=pt});
  });
  return t;
}

// Draft logic
function currentTeamIdx(){
  const n=state.draftOrder.length,pick=state.currentPick;
  if(n===0)return-1;
  const posInRound=pick%(n*2);
  return posInRound<n?posInRound:(2*n-1-posInRound);
}
function currentTeam(){
  const n=state.draftOrder.length;if(!n)return null;
  const idx=currentTeamIdx();if(idx<0)return null;
  const team=state.draftOrder[idx];
  if((state.picks[team]||[]).length>=4){state.currentPick++;save();return currentTeam()}
  return team;
}
function save(){syncPush()}

function startLottery(){
  if(state.signups.length<2){toast('Need at least 2 players!');return}
  const shuffled=[...state.signups];
  for(let i=shuffled.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[shuffled[i],shuffled[j]]=[shuffled[j],shuffled[i]]}
  state.draftOrder=shuffled;state.phase='lottery';save();render();animateLottery();
}

function animateLottery(){
  const c=$('lottery-container'),r=$('lottery-revealed');
  c.innerHTML='<div class="lottery-spinner"><span class="dice">🎲</span> Shuffling...</div>';r.classList.add('hidden');
  setTimeout(()=>{
    c.innerHTML='';
    state.draftOrder.forEach((name,i)=>{
      const s=document.createElement('div');s.className='lottery-slot';
      s.innerHTML=`<span class="lottery-pos">${i+1}</span><span class="lottery-name">${esc(name)}</span>`;
      c.appendChild(s);setTimeout(()=>s.classList.add('revealed'),i*500+200);
    });
    setTimeout(()=>{r.classList.remove('hidden');toast('Order revealed!')},state.draftOrder.length*500+600);
  },1500);
}

function beginDraft(){
  state.picks={};state.currentPick=0;state.phase='draft';save();render();toast('Draft begins!');
}

function makePick(){
  const team=currentTeam();if(!team||!selectedPlayer)return;
  if(team!==localName){toast('Not your turn!');return}
  if(!state.picks[team])state.picks[team]=[];
  state.picks[team].push(selectedPlayer);
  state.currentPick++;selectedPlayer=null;
  const total=state.draftOrder.length*4,made=Object.values(state.picks).flat().length;
  if(made>=total){state.phase='leaderboard';toast('Draft complete!')}
  save();render();
}

const CORS=u=>`https://api.allorigins.win/raw?url=${encodeURIComponent(u)}`;
const AFL_SEASON='CD_S2026014';
async function fetchVotes(){
  const el=$('vote-status');el.textContent='Fetching votes...';
  try{
    const urls=[`https://api.afl.com.au/cfs/afl/brownlow/leaderboard?compSeasonId=${AFL_SEASON}`,`https://api.afl.com.au/cfs/afl/brownlow/predictor?compSeasonId=${AFL_SEASON}`];
    for(const u of urls){try{const r=await fetch(CORS(u),{signal:AbortSignal.timeout(10000)});if(r.ok){const d=await r.json();if(parseAfl(d)){state.lastVoteFetch=Date.now();el.textContent='✓ Updated';save();renderLeaderboard();return}}}catch{}}
    try{const r=await fetch('https://api.squiggle.com.au/?q=brownlow;year=2026',{signal:AbortSignal.timeout(8000)});const t=await r.text();const lines=t.split('\n').filter(l=>l.trim());if(lines.length<2){el.textContent='⚠ No vote data yet';return}
    const hd=lines[0].split('\t').map(h=>h.trim().toLowerCase());const v={};for(let i=1;i<lines.length;i++){const vals=lines[i].split('\t');const row={};hd.forEach((h,j)=>row[h]=vals[j]?.trim());if(row.votes&&row.player&&row.round){const rk='R'+row.round;if(!v[rk])v[rk]={};v[rk][row.player]=parseInt(row.votes)||0}}
    if(Object.keys(v).length){state.votes=v;state.lastVoteFetch=Date.now();el.textContent='✓ Updated'}else el.textContent='⚠ No vote data yet'}catch{el.textContent='⚠ Couldn\'t fetch votes'}
  }catch{el.textContent='⚠ Couldn\'t fetch votes'}
  save();renderLeaderboard();
}
function parseAfl(d){let f=false;const a=Array.isArray(d)?d:[d];a.forEach(e=>{const rk='R'+(e.roundNumber||e.round||'');if(!state.votes[rk])state.votes[rk]={};(e.players||[]).forEach(p=>{state.votes[rk][p.name||p.playerName]=p.votes||p.voteCount||0;f=true})});return f}
